message("----- KisakCOD Client(Vita) -----")

project(KisakCOD-vita)

# Include all common files between mp/sp/dedi
include(${SCRIPTS_DIR}/common_files.cmake)

# The Vita build is the single-player engine
include(${SCRIPTS_DIR}/sp/sp_files.cmake)

include(${CMAKE_CURRENT_SOURCE_DIR}/vita_files.cmake)

include(${PLATFORM_DIR}/${KISAK_PLATFORM}/platform.cmake)

# vita_create_self reads from the target's own binary dir, the root sends binaries to bin/
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})

# excluded: WIN32_SRC, SOUND, SOUND_BACKEND, BINKLIB, MSSLIB, SPEEX, STEAM_HEADERS, src/radiant
add_executable(${PROJECT_NAME}
    ${SRCROOT}
    ${AIM_ASSIST}
    ${BGAME}
    ${CGAME}
    ${CLIENT}
    ${COMMON}
    ${DATABASE}
    ${DEVGUI}
    ${DYNENTITY}
    ${EFFECTSCORE}
    ${GAME}
    ${GFX_D3D}
    ${ODE}
    ${PHYSICS}
    ${PHYSICS_ODE}
    ${QCOMMON}
    ${RAGDOLL}
    ${SCRIPT}
    ${SERVER}
    ${STRINGED}
    ${UI}
    ${UNIVERSAL}
    ${XANIM}
    ${VITA_ZLIB}
    ${VITA_PLATFORM}
    ${VITA_GXM}
    ${VITA_INPUT}
)

# CINEMA stays off: deps/binklib is not linked here
target_compile_definitions(${PROJECT_NAME} PRIVATE KISAK_VITA KISAK_SP)

target_include_directories(${PROJECT_NAME} PRIVATE ${SRC_DIR} ${DEPS_DIR})

# vita_prefix.h uses extern "C", so the zlib and ode C files must not receive it
target_compile_options(${PROJECT_NAME} PRIVATE
    "$<$<COMPILE_LANGUAGE:CXX>:-include;${SRC_DIR}/vita/platform/vita_prefix.h>"
)

target_link_libraries(${PROJECT_NAME} ${VITA_STUB_LIBS})

add_dependencies(${PROJECT_NAME} update_build_number)

set(VITA_APP_NAME "Call of Duty 4 Vita")
set(VITA_TITLEID  "KCOD00003")
set(VITA_VERSION  "01.00")

# without ATTRIBUTE2 the system ignores the livearea files and shows the default gate
set(VITA_MKSFOEX_FLAGS "-d ATTRIBUTE2=12" CACHE STRING "vita-mksfoex flags" FORCE)

vita_create_self(${PROJECT_NAME}.self ${PROJECT_NAME} UNSAFE)

set(SCE_SYS ${CMAKE_SOURCE_DIR}/sce_sys)
set(PORT_VERSION ${VITA_VERSION})
configure_file(
    ${SCE_SYS}/livearea/contents/template.xml.in
    ${CMAKE_CURRENT_BINARY_DIR}/generated/livearea/template.xml
    @ONLY)

# the artwork is generated by scripts/vita/make_livearea.py and is not in the repo
if(EXISTS ${SCE_SYS}/icon0.png AND EXISTS ${SCE_SYS}/livearea/contents/bg.png
   AND EXISTS ${SCE_SYS}/livearea/contents/startup.png)
  set(LIVEAREA_FILES
    FILE ${SCE_SYS}/icon0.png sce_sys/icon0.png
    FILE ${SCE_SYS}/livearea/contents/bg.png sce_sys/livearea/contents/bg.png
    FILE ${SCE_SYS}/livearea/contents/startup.png sce_sys/livearea/contents/startup.png
    FILE ${CMAKE_CURRENT_BINARY_DIR}/generated/livearea/template.xml sce_sys/livearea/contents/template.xml
  )
else()
  message(STATUS "livearea artwork absent; run scripts/vita/make_livearea.py to add it")
  set(LIVEAREA_FILES "")
endif()

vita_create_vpk(${PROJECT_NAME}.vpk ${VITA_TITLEID} ${PROJECT_NAME}.self
    VERSION ${VITA_VERSION}
    NAME ${VITA_APP_NAME}
    ${LIVEAREA_FILES}
)
